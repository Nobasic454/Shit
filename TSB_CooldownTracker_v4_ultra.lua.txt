-- Linear colour lerp (component-wise in linear RGB space)
local function lerpColor(a, b, t)
    return Color3.new(
        a.R + (b.R - a.R) * t,
        a.G + (b.G - a.G) * t,
        a.B + (b.B - a.B) * t
    )
end

-- Creates a rounded drop-shadow frame parented under `parent`
local function makeShadow(parent, zIndex)
    local shadow = Instance.new("Frame")
    shadow.Name                   = "DropShadow"
    shadow.BackgroundColor3       = BLACK
    shadow.BackgroundTransparency = 0.65
    shadow.BorderSizePixel        = 0
    shadow.ZIndex                 = zIndex or 1
    shadow.Parent                 = parent
    local sc  = Instance.new("UICorner", shadow)
    sc.CornerRadius = UDim.new(0, 10)
    local sg  = Instance.new("UIGradient", shadow)
    sg.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0,    0.48),
        NumberSequenceKeypoint.new(0.45, 0.62),
        NumberSequenceKeypoint.new(1,    0.84),
    })
    sg.Rotation = 90
    return shadow
end

-- Builds the repeating enchant-sweep tween loop on a sweep strip frame.
-- `delay` staggers bars so they don't all sweep simultaneously.
local function startEnchantSweep(sweep, delay)
    task.spawn(function()
        if delay and delay > 0 then task.wait(delay) end
        while sweep and sweep.Parent do
            local sweepPct = (Options.SweepW  and Options.SweepW.Value  or 40) / 100
            local sweepDur = (Options.SweepDur and Options.SweepDur.Value or 2.5)
            local sweepGap = (Options.SweepGap and Options.SweepGap.Value or 3.4)
            -- reset to the left
            pcall(function()
                sweep.Size     = UDim2.new(sweepPct, 0, 1.2, 0)
                sweep.Position = UDim2.new(-sweepPct - 0.05, 0, -0.1, 0)
            end)
            if Toggles.SweepOn and Toggles.SweepOn.Value then
                local t = TweenService:Create(sweep,
                    TweenInfo.new(sweepDur, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut),
                    { Position = UDim2.new(1.15, 0, -0.1, 0) })
                t:Play()
                task.wait(sweepGap)
            else
                task.wait(1.0)
            end
        end
    end)
end

-- Blinks a tiny sparkle dot inside a fill bar.
local function startSparkle(sp, delay)
    task.spawn(function()
        if delay and delay > 0 then task.wait(delay) end
        while sp and sp.Parent do
            task.wait(math.random(8, 30) * 0.1)
            if not (sp and sp.Parent) then break end
            sp.BackgroundTransparency = 0.04
            task.wait(0.055)
            sp.BackgroundTransparency = 0.40
            task.wait(0.055)
            sp.BackgroundTransparency = 0.04
            TweenService:Create(sp,
                TweenInfo.new(0.72, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                { BackgroundTransparency = 1 }):Play()
        end
    end)
end

-- ████████████████████████████████████████████████████████████████████████████
--  PLAYER BAR CONSTRUCTION
--  Each ability gets a layered frame stack:
--   drop shadow → outer halo glow → inner glow → bar background
--   → fill clip → fill bar → sweep + sparkles + shine
--   → label above, number overlaid, corner gems at ends
-- ████████████████████████████████████████████████████████████████████████████

local bars = {}

local function createBar(name, barIdx)
    local cfg = ABILITY_CFG[name]

    -- ── Root container (transparent, for positioning) ─────────────────────────
    local container = Instance.new("Frame")
    container.Name                   = name .. "_Container"
    container.BackgroundTransparency = 1
    container.BorderSizePixel        = 0
    container.ZIndex                 = 2
    container.Parent                 = screenGui

    -- ── Drop shadow ───────────────────────────────────────────────────────────

-- ████████████████████████████████████████████████████████████████████████████
--  EFFECTS: PULSE  (bar pops on trigger)
-- ████████████████████████████████████████████████████████████████████████████

local function doPulse(barData)
    if not Toggles.PulseOn.Value then return end
    if barData.pulseActive then return end
    barData.pulseActive = true
    local frame   = barData.frame
    local scale   = Options.PulseSc.Value  or 1.07
    local halfDur = (Options.PulseDur.Value or 0.38) * 0.5
    local orig    = frame.Size
    local scaled  = UDim2.new(
        orig.X.Scale * scale, orig.X.Offset * scale,
        orig.Y.Scale * scale, orig.Y.Offset * scale)
    TweenService:Create(frame,
        TweenInfo.new(halfDur, Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
        { Size = scaled }):Play()
    task.delay(halfDur, function()
        TweenService:Create(frame,
            TweenInfo.new(halfDur, Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
            { Size = orig }):Play()
        task.delay(halfDur + 0.05, function()
            barData.pulseActive = false
        end)
    end)
end

-- ████████████████████████████████████████████████████████████████████████████
--  EFFECTS: READY FLASH  (fill colour flickers on ready)
-- ████████████████████████████████████████████████████████████████████████████

local function doReadyFlash(barData)
    if not Toggles.FlashOn.Value then return end
    if barData.readyFlashing then return end
    barData.readyFlashing = true
    local fill     = barData.fill
    local flashClr = (Options.FlashClr and Options.FlashClr.Value) or Color3.fromRGB(255, 255, 100)
    local origClr  = fill.BackgroundColor3
    local count    = Options.FlashCnt.Value or 3
    local spd      = Options.FlashSpd.Value or 0.14
    local function flashN(n)
        if n <= 0 then
            pcall(function() fill.BackgroundColor3 = origClr end)
            barData.readyFlashing = false
            return
        end
        TweenService:Create(fill,
            TweenInfo.new(spd, Enum.EasingStyle.Linear),
            { BackgroundColor3 = flashClr }):Play()
        task.delay(spd, function()
            TweenService:Create(fill,
                TweenInfo.new(spd, Enum.EasingStyle.Linear),
                { BackgroundColor3 = origClr }):Play()
            task.delay(spd, function() flashN(n - 1) end)
        end)
    end
    task.spawn(flashN, count)
end

-- ████████████████████████████████████████████████████████████████████████████
--  EFFECTS: READY BREATH  (border + glows breathe rhythmically while ready)
-- ████████████████████████████████████████████████████████████████████████████

local function doReadyPulse(barData)
    if barData.readyPulseActive then return end
    barData.readyPulseActive = true
    local stroke    = barData.barStroke
    local glow      = barData.glow
    local outerGlow = barData.outerGlow
    local function pulse()
        if not barData.readyPulseActive then return end
        local breathSpd = (Options.BreathSpd and Options.BreathSpd.Value) or 1.0
        local bMin      = (Options.BreathMin and Options.BreathMin.Value) or 0.56
        local bMax      = (Options.BreathMax and Options.BreathMax.Value) or 0.0
        if not (Toggles.BreathOn and Toggles.BreathOn.Value) then
            task.wait(0.5)
            if barData.readyPulseActive then task.spawn(pulse) end
            return
        end
        -- breathe in (bright)
        TweenService:Create(stroke,
            TweenInfo.new(breathSpd, Enum.EasingStyle.Sine),
            { Transparency = bMax }):Play()
        TweenService:Create(glow,
            TweenInfo.new(breathSpd, Enum.EasingStyle.Sine),
            { BackgroundTransparency = 0.42 }):Play()
        TweenService:Create(outerGlow,
            TweenInfo.new(breathSpd, Enum.EasingStyle.Sine),
            { BackgroundTransparency = 0.70 }):Play()
        task.wait(breathSpd)
        if not barData.readyPulseActive then return end
        -- breathe out (dim)
        TweenService:Create(stroke,
            TweenInfo.new(breathSpd, Enum.EasingStyle.Sine),
            { Transparency = bMin }):Play()
        TweenService:Create(glow,
            TweenInfo.new(breathSpd, Enum.EasingStyle.Sine),
            { BackgroundTransparency = 0.86 }):Play()
        TweenService:Create(outerGlow,
            TweenInfo.new(breathSpd, Enum.EasingStyle.Sine),
            { BackgroundTransparency = 0.96 }):Play()
        task.wait(breathSpd)
        if barData.readyPulseActive then task.spawn(pulse) end
    end
    task.spawn(pulse)
end

local function stopReadyPulse(barData)
    barData.readyPulseActive = false
end

-- ████████████████████████████████████████████████████████████████████████████
--  MAIN PLAYER BAR UPDATE LOOP
-- ████████████████████████████████████████████████████████████████████████████

RunService.Heartbeat:Connect(function(dt)
    local width      = Options.Width.Value
    local height     = Options.Height.Value
    local spacing    = Options.Spacing.Value
    local cornerR    = Options.Corner.Value
    local posX       = Options.PosX.Value
    local posY       = Options.PosY.Value
    local bgTransp   = Options.BGTransp.Value
    local fillTransp = Options.FillTransp.Value
    local textSize   = Options.TextSize.Value
    local lblSize    = Options.LabelSize.Value
    local lerpSpd    = Options.FillSpeed.Value
    local fillDir    = Options.FillDir.Value
    local glowSz     = Options.GlowSize.Value
    local shadowOff  = Options.ShadowOff.Value
    local shadowOp   = Options.ShadowOp.Value
    local gemSz      = Options.GemSize.Value
    local font       = FONT_MAP[Options.FontChoice.Value] or Enum.Font.GothamBold

    local showBars    = Toggles.ShowBars.Value
    local showNums    = Toggles.ShowNumbers.Value
    local showLabels  = Toggles.ShowLabels.Value
    local showReady   = Toggles.ShowReady.Value
    local showPct     = Toggles.ShowPercent.Value
    local showBorder  = Toggles.ShowBorder.Value
    local showShine   = Toggles.ShowShine.Value
    local showGlow    = Toggles.ShowGlow.Value
    local showShadow  = Toggles.ShowShadow.Value
    local showGems    = Toggles.ShowGems.Value
    local colorByTime = Toggles.ColorByTime.Value

    local totalW = #ABILITY_ORDER * width + (#ABILITY_ORDER - 1) * spacing
    local startX = -totalW / 2

    for i, name in ipairs(ABILITY_ORDER) do
        local data = bars[name]
        local cfg  = ABILITY_CFG[name]

        -- Tick down the cooldown timer
        if data.time > 0 then
            data.time = math.max(data.time - dt, 0)
        end

        -- Lerped fill ratio: 0 = empty (on cd), 1 = full (ready)
        local realRatio = 1 - (data.time / math.max(data.duration, 0.001))
        local lf        = math.min(lerpSpd * dt, 1)
        data.visRatio   = data.visRatio + (realRatio - data.visRatio) * lf

        local isOnCd = data.time > 0
        local ratio  = math.clamp(data.visRatio, 0, 1)
        local hasFill = ratio > 0.03

        -- Transition: just became ready → flash + breath pulse
        -- Transition: just triggered  → stop breath, do pop pulse
        if data.prevOnCd and not isOnCd then
            task.spawn(doReadyFlash, data)
            task.spawn(doReadyPulse, data)
        elseif not data.prevOnCd and isOnCd then
            stopReadyPulse(data)
        end
        data.prevOnCd = isOnCd

        -- ── Positioning ────────────────────────────────────────────────────────
        local xOff = startX + (i - 1) * (width + spacing)
        local lblH  = showLabels and (lblSize + 3) or 0
        local totH  = height + lblH + (showLabels and 2 or 0)

        data.container.Size     = UDim2.new(0, width, 0, totH)
        data.container.Position = UDim2.new(posX, xOff, posY, -totH / 2)

        -- ── Drop shadow ────────────────────────────────────────────────────────
        data.shadow.Size               = UDim2.new(1, shadowOff * 2, 0, height + shadowOff * 2)
        data.shadow.Position           = UDim2.new(0, -shadowOff, 0, lblH + shadowOff)
        data.shadow.BackgroundTransparency = (showShadow and showBars) and shadowOp or 1

        -- ── Glow layers ────────────────────────────────────────────────────────
        local glowClr = Options[cfg.glowKey].Value
        data.outerGlow.Size             = UDim2.new(0, width + glowSz * 2, 0, height + glowSz)
        data.outerGlow.Position         = UDim2.new(0, -glowSz, 0, lblH - glowSz * 0.4)
        data.outerGlow.BackgroundColor3 = glowClr
        data.outerGlow.Visible          = showGlow and hasFill and not data.readyPulseActive

        data.glow.Size             = UDim2.new(0, width + 10, 0, height + 8)
        data.glow.Position         = UDim2.new(0, -5, 0, lblH - 3)
        data.glow.BackgroundColor3 = glowClr
        data.glow.Visible          = showGlow and hasFill and not data.readyPulseActive

        -- ── Ability label ──────────────────────────────────────────────────────
        data.abilityLabel.Size       = UDim2.new(1, 0, 0, lblH)
        data.abilityLabel.Position   = UDim2.new(0, 2, 0, 0)
        data.abilityLabel.TextSize   = lblSize
        data.abilityLabel.Font       = font
        data.abilityLabel.TextColor3 = Options[cfg.lblKey].Value
        data.abilityLabel.Visible    = showLabels

        -- ── Colour resolution + dynamic danger tint ────────────────────────────
        local fillClr   = Options[cfg.fillKey].Value
        local dangerClr = (Options[cfg.dangerKey] and Options[cfg.dangerKey].Value) or DANGER_C
        local bgClr     = Options[cfg.bgKey].Value
        local numClr    = Options[cfg.numKey].Value
        local borderClr = Options[cfg.borderKey].Value

        -- cdProgress: 1 = ability just used, 0 = about to be ready
        local cdProgress = 1 - realRatio
        local dynamicFill = fillClr
        if colorByTime and isOnCd and cdProgress > 0.55 then
            local t = (cdProgress - 0.55) / 0.45
            dynamicFill = lerpColor(fillClr, dangerClr, math.min(t * 0.65, 0.65))
        end

        -- ── Bar frame sizing / corners ─────────────────────────────────────────
        data.frame.Size     = UDim2.new(0, width, 0, height)
        data.frame.Position = UDim2.new(0, 0, 0, lblH + (showLabels and 2 or 0))
        data.barCorner.CornerRadius  = UDim.new(0, cornerR)
        data.fillCorner.CornerRadius = UDim.new(0, cornerR)

        -- ── Border ─────────────────────────────────────────────────────────────
        data.barStroke.Enabled = showBorder
        data.barStroke.Color   = borderClr
        if not data.readyPulseActive then
            data.barStroke.Transparency = 0.10
        end

        -- ── Corner gems ────────────────────────────────────────────────────────
        local gemH = math.min(gemSz, height - 2)
        data.gemLeft.Size                   = UDim2.new(0, gemSz, 0, gemH)
        data.gemLeft.Position               = UDim2.new(0, gemSz * 0.5, 0.5, 0)
        data.gemLeft.BackgroundColor3       = dynamicFill
        data.gemLeft.BackgroundTransparency = (showGems and hasFill) and 0.15 or 1

        data.gemRight.Size                   = UDim2.new(0, gemSz, 0, gemH)
        data.gemRight.Position               = UDim2.new(1, -(gemSz * 1.5), 0.5, 0)
        data.gemRight.BackgroundColor3       = dynamicFill
        data.gemRight.BackgroundTransparency = (showGems and hasFill) and 0.15 or 1

        -- ── Shine ──────────────────────────────────────────────────────────────
        data.shine.Visible     = showShine
        data.shineLine.Visible = showShine

        -- ── Sparkle tint ───────────────────────────────────────────────────────
        for _, sp in ipairs(data.sparkles) do
            sp.BackgroundColor3 = dynamicFill
        end

        -- ── Visibility + fill ──────────────────────────────────────────────────
        if showBars then
            data.container.Visible            = true
            data.frame.Visible                = true
            data.fill.Visible                 = true
            data.frame.BackgroundColor3       = bgClr
            data.frame.BackgroundTransparency = bgTransp
            data.fill.BackgroundColor3        = dynamicFill
            data.fill.BackgroundTransparency  = fillTransp
            if fillDir == "Left" then
                data.fill.AnchorPoint = Vector2.new(0, 0)
                data.fill.Position    = UDim2.new(0, 0, 0, 0)
            else
                data.fill.AnchorPoint = Vector2.new(1, 0)
                data.fill.Position    = UDim2.new(1, 0, 0, 0)
            end
            data.fill.Size = UDim2.new(ratio, 0, 1, 0)
        elseif showNums then
            data.container.Visible            = true
            data.frame.Visible                = true
            data.fill.Visible                 = false
            data.frame.BackgroundTransparency = 1
        else
            data.container.Visible = false
        end

        -- ── Text ───────────────────────────────────────────────────────────────
        if data.container.Visible then
            data.text.Font     = font
            data.text.TextSize = textSize
            if showNums then
                if isOnCd then
                    data.text.TextColor3 = numClr
                    data.text.Text = showPct
                        and string.format("%.0f%%", realRatio * 100)
                        or  string.format("%.1f",   data.time)
                elseif showReady then
                    data.text.Text       = "✦ READY ✦"
                    data.text.TextColor3 = READY_C
                else
                    data.text.Text = ""
                end
            else
                data.text.Text = ""
            end
        end
    end
end)

local ANIM_MAP = {
    { name = "Dash", ids = { 10479335397, 10491993682 } },
    { name = "Side", ids = { 10480793962, 10480796021 } },
}

local function trigger(name)
    local data = bars[name]
    if not data then return end
    data.time     = data.duration
    data.prevOnCd = true
    stopReadyPulse(data)
    task.spawn(doPulse, data)
end

local function onLocalAnimId(id)
    for _, entry in ipairs(ANIM_MAP) do
        for _, animId in ipairs(entry.ids) do
            if id == animId then trigger(entry.name) end
        end
    end
end

local function detectLocal()
    local char     = player.Character or player.CharacterAdded:Wait()
    local hum      = char:WaitForChild("Humanoid", 10)
    local animator = hum and hum:WaitForChild("Animator", 10)
    if not animator then return end
    animator.AnimationPlayed:Connect(function(track)
        local id = tonumber(track.Animation.AnimationId:match("%d+"))
        if id then onLocalAnimId(id) end
    end)
end

player.CharacterAdded:Connect(function() task.spawn(detectLocal) end)
if player.Character then task.spawn(detectLocal) end

-- Sweep animation for enemy bars (slightly shorter cadence than player)
local function startEnemySweep(sweep, delay)
    task.spawn(function()
        if delay and delay > 0 then task.wait(delay) end
        while sweep and sweep.Parent do
            local sweepDur = (Options.SweepDur and Options.SweepDur.Value or 2.5) * 0.72
            local sweepGap = (Options.SweepGap and Options.SweepGap.Value or 3.4) * 0.80
            pcall(function()
                sweep.Position = UDim2.new(-0.48, 0, -0.1, 0)
            end)
            if Toggles.SweepOn and Toggles.SweepOn.Value then
                local t = TweenService:Create(sweep,
                    TweenInfo.new(sweepDur, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut),
                    { Position = UDim2.new(1.18, 0, -0.1, 0) })
                t:Play()
                task.wait(sweepGap)
            else
                task.wait(1.0)
            end
        end
    end)
end

-- Sparkle for enemy bars (3 dots, subtler)
local function startEnemySparkle(sp, delay)
    task.spawn(function()
        if delay and delay > 0 then task.wait(delay) end
        while sp and sp.Parent do
            task.wait(math.random(12, 38) * 0.1)
            if not (sp and sp.Parent) then break end
            sp.BackgroundTransparency = 0.10
            task.wait(0.06)
            sp.BackgroundTransparency = 0.50
            task.wait(0.06)
            sp.BackgroundTransparency = 0.10
            TweenService:Create(sp,
                TweenInfo.new(0.65, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                { BackgroundTransparency = 1 }):Play()
        end
    end)
end

    end

-- ████████████████████████████████████████████████████████████████████████████
--  ENEMY OVERHEAD: TRIGGER + SETUP + CLEANUP
-- ████████████████████████████████████████████████████████████████████████████

local function triggerOverhead(p, abilityName)
    local cfg = OH_CFG[abilityName]
    if not cfg then return end
    if not playerTrackers[p] then playerTrackers[p] = {} end
    if not playerTrackers[p][abilityName] then
        playerTrackers[p][abilityName] = { time = 0, visRatio = 0 }
    end
    playerTrackers[p][abilityName].time = cfg.defaultCD
end

-- Animation IDs for enemy detection (Front Dash + Side Dash)
local ENEMY_ANIM_MAP = {
    { name = "Dash", ids = { 10479335397, 10491993682 } },
    { name = "Side", ids = { 10480793962, 10480796021 } },
}

local function setupPlayer(p)
    if p == player then return end

    playerTrackers[p] = {}
    for abilityName in pairs(OH_CFG) do
        playerTrackers[p][abilityName] = { time = 0, visRatio = 0 }
    end

    local function onChar(char)
        char:WaitForChild("Head", 10)
        task.wait(0.12)
        buildOverhead(p)

        local hum  = char:WaitForChild("Humanoid", 10)
        local anim = hum and hum:WaitForChild("Animator", 10)
        if not anim then return end

        anim.AnimationPlayed:Connect(function(track)
            local id = tonumber(track.Animation.AnimationId:match("%d+"))
            if not id then return end
            for _, entry in ipairs(ENEMY_ANIM_MAP) do
                for _, animId in ipairs(entry.ids) do
                    if id == animId then
                        triggerOverhead(p, entry.name)
                    end
                end
            end
        end)
    end

    if p.Character then task.spawn(onChar, p.Character) end
    p.CharacterAdded:Connect(onChar)
end

-- Initialise all already-present players and listen for new joins/leaves
for _, p in ipairs(Players:GetPlayers()) do
    setupPlayer(p)
end
Players.PlayerAdded:Connect(setupPlayer)

Players.PlayerRemoving:Connect(function(p)
    local tracker = playerTrackers[p]
    if not tracker then return end
    if tracker._billboard then
        pcall(function() tracker._billboard:Destroy() end)
    end
    playerTrackers[p] = nil
end)

-- ████████████████████████████████████████████████████████████████████████████
--  EVASIVE DETECTION  (local player + enemies via RagdollCancel)
-- ████████████████████████████████████████████████████████████████████████████

LiveFolder.DescendantAdded:Connect(function(child)
    if child.Name ~= "RagdollCancel" then return end
    local char = child.Parent
    if char == player.Character then
        trigger("Evasive")
        return
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character == char then
            triggerOverhead(p, "Evasive")
            break
        end
    end
end)

--  ENEMY OVERHEAD BAR UPDATE LOOP
-- ████████████████████████████████████████████████████████████████████████████

RunService.Heartbeat:Connect(function(dt)
    local showEnemy    = Toggles.ShowEnemy.Value
    local onlyCd       = Toggles.EnemyOnlyCd.Value
    local showFill     = Toggles.EnemyShowFill.Value
    local showNum      = Toggles.EnemyShowNum.Value
    local showLbl      = Toggles.EnemyShowLbl.Value
    local showGlow     = Toggles.EnemyShowGlow.Value
    local showBord     = Toggles.EnemyShowBord.Value
    local showShine    = Toggles.EnemyShowShine.Value
    local showGems     = Toggles.EnemyShowGems.Value
    local ohAlTop      = Toggles.EnemyAlTop.Value
    local ohW          = Options.EnemyW.Value
    local ohH          = Options.EnemyH.Value
    local ohSp         = Options.EnemySp.Value
    local lblSz        = Options.EnemyLblSz.Value
    local ohSO         = Options.EnemyStuds.Value
    local ohBGT        = Options.EnemyBGT.Value
    local ohTxtSz      = Options.EnemyTxtSz.Value
    local cornerR      = Options.EnemyCorner.Value
    local strokeOp     = Options.EnemyStrokeOp.Value
    local lerpSpd      = Options.EnemyLerpSpd.Value
    local fillDir      = Options.EnemyFillDir.Value
    local glowAlpha    = Options.EnemyGlowAlpha.Value
    local gemSzEnemy   = Options.EnemyGemSz.Value
    local ohFont       = FONT_MAP[Options.EnemyFont.Value] or Enum.Font.GothamBold
    local lblH         = lblSz + 2
    local entryH       = ohH + lblH + 3
    local totH         = #OH_ORDER * entryH + (#OH_ORDER - 1) * ohSp

    for p, tracker in pairs(playerTrackers) do
        -- Update billboard-level properties
        local bill = tracker._billboard
        if bill then
            bill.Enabled     = showEnemy
            bill.AlwaysOnTop = ohAlTop
            bill.StudsOffset = Vector3.new(0, ohSO, 0)
            bill.Size        = UDim2.new(0, ohW, 0, totH)
        end

        -- Per-ability bar update
        for abilityName, data in pairs(tracker) do
            if abilityName == "_billboard" then continue end
            local cfg = OH_CFG[abilityName]
            if not cfg then continue end

            -- Advance cooldown timer
            if data.time > 0 then
                data.time = math.max(data.time - dt, 0)
            end

            -- Lerp visual fill ratio
            if not data.visRatio then data.visRatio = 0 end
            local realRatio = 1 - (data.time / math.max(cfg.defaultCD, 0.001))
            local lf = math.min(lerpSpd * dt, 1)
            data.visRatio = data.visRatio + (realRatio - data.visRatio) * lf

            local isOnCd  = data.time > 0
            local ratio   = math.clamp(data.visRatio, 0, 1)
            local hasFill = ratio > 0.03

            -- Resolve UI references (may be absent if billboard not yet built)
            local entry     = data.entry
            local outerGlow = data.outerGlow
            local lbl       = data.lbl
            local bgFrame   = data.bgFrame
            local bgStroke  = data.bgStroke
            local bgCorner  = data.bgCorner
            local fillBar   = data.fillBar
            local fillCorner= data.fillCorner
            local sparkles  = data.sparkles
            local shineLine = data.shineLine
            local shineOver = data.shineOver
            local gemLeft   = data.gemLeft
            local gemRight  = data.gemRight
            local numLbl    = data.numLbl

            if not entry or not entry.Parent then continue end

            -- Colour lookups
            local fillClr   = Options[cfg.fillKey].Value
            local bgClr     = Options[cfg.bgKey].Value
            local borderClr = Options[cfg.borderKey].Value
            local numClr    = Options[cfg.numKey].Value
            local lblClr    = Options[cfg.lblKey].Value
            local glowClr   = Options[cfg.glowKey].Value

            -- Show/hide entire entry row
            local shouldShow = showEnemy and (isOnCd or not onlyCd)
            entry.Visible = shouldShow
            if not shouldShow then continue end

            -- ── Outer glow ─────────────────────────────────────────────────────
            if outerGlow then
                outerGlow.BackgroundColor3       = glowClr
                outerGlow.BackgroundTransparency = (showGlow and hasFill)
                    and glowAlpha or 1
            end

            -- ── Label ──────────────────────────────────────────────────────────
            if lbl then
                lbl.Visible    = showLbl
                lbl.TextColor3 = lblClr
                lbl.Font       = ohFont
                lbl.TextSize   = lblSz
                lbl.TextStrokeTransparency = strokeOp
            end

            -- ── BG frame ───────────────────────────────────────────────────────
            if bgFrame then
                bgFrame.BackgroundColor3       = bgClr
                bgFrame.BackgroundTransparency = ohBGT
            end
            if bgCorner then
                bgCorner.CornerRadius = UDim.new(0, cornerR)
            end

            -- ── Border stroke ──────────────────────────────────────────────────
            if bgStroke then
                bgStroke.Enabled      = showBord
                bgStroke.Color        = borderClr
                bgStroke.Transparency = 0.16
            end

            -- ── Fill bar ───────────────────────────────────────────────────────
            if fillBar then
                fillBar.Visible             = showFill
                fillBar.BackgroundColor3    = fillClr
                fillBar.BackgroundTransparency = 0
                if fillCorner then
                    fillCorner.CornerRadius = UDim.new(0, cornerR)
                end
                if fillDir == "Left" then
                    fillBar.AnchorPoint = Vector2.new(0, 0)
                    fillBar.Position    = UDim2.new(0, 0, 0, 0)
                else
                    fillBar.AnchorPoint = Vector2.new(1, 0)
                    fillBar.Position    = UDim2.new(1, 0, 0, 0)
                end
                fillBar.Size = UDim2.new(ratio, 0, 1, 0)
            end

            -- ── Sparkle tint ───────────────────────────────────────────────────
            if sparkles then
                for _, sp in ipairs(sparkles) do
                    sp.BackgroundColor3 = fillClr
                end
            end

            -- ── Shine ──────────────────────────────────────────────────────────
            if shineLine then shineLine.Visible = showShine end
            if shineOver  then shineOver.Visible  = showShine end

            -- ── Corner gems ────────────────────────────────────────────────────
            if gemLeft then
                gemLeft.Size                   = UDim2.new(0, gemSzEnemy, 0, gemSzEnemy)
                gemLeft.BackgroundColor3       = fillClr
                gemLeft.BackgroundTransparency = (showGems and hasFill) and 0.18 or 1
            end
            if gemRight then
                gemRight.Size                   = UDim2.new(0, gemSzEnemy, 0, gemSzEnemy)
                gemRight.BackgroundColor3       = fillClr
                gemRight.BackgroundTransparency = (showGems and hasFill) and 0.18 or 1
            end

            -- ── Number text ────────────────────────────────────────────────────
            if numLbl then
                numLbl.Visible    = showNum
                numLbl.Font       = ohFont
                numLbl.TextSize   = ohTxtSz
                numLbl.TextStrokeTransparency = strokeOp
                if showNum then
                    if isOnCd then
                        numLbl.TextColor3 = numClr
                        numLbl.Text = string.format("%.1f", data.time)
                    else
                        numLbl.TextColor3 = READY_C
                        numLbl.Text = "READY"
                    end
                else
                    numLbl.Text = ""
                end
            end
        end
    end
end)
