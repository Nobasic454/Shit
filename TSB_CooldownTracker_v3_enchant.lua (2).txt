--[[
    The Cooldown Tracker v4 is designed to run in highly dynamic Roblox environments with zero
    frame drops, premium graphics, and extremely precise tracking. Below is a breakdown of the 
    core architectural modules included in this single-file distribution:

    1. DESIGN SYSTEM & TOKEN MANAGEMENT:
       - Structured with strict color palettes, layouts, and style scales.
       - A built-in "Theme Manager" controls rapid styling switches dynamically. Themes act as
         global state controllers, automatically updating both local UI and enemy overhead elements.

    2. OPTIMIZED UI PARTICLE PHYSICS SIMULATION (LumeParticleEngine):
       - A customized particle simulation library written from scratch.
       - Solves the performance bottleneck of Roblox GUI creation by implementing a custom "ObjectPool".
       - Recycles visual particle frames to avoid garbage collection spikes (major cause of micro-stuttering).
       - Features: Dynamic velocity, wind vector, gravity, angular velocity, and scale transitions.
       - Math derivation:
         Position(t) = Position(t-dt) + Velocity * dt
         Velocity(t) = Velocity(t-dt) + (Gravity + Wind)*dt - (Drag * Velocity * dt)
         Opacity(t)  = Lerp(InitialOpacity, EndOpacity, ElapsedTime / Lifetime)

    3. ENEMY OVERHEAD RENDERING ENGINE (BillboardManager):
       - Dynamically builds and updates custom BillboardGuis on the heads of nearby players.
       - Replaces standard TextLabels with full-fledged, high-end progress bars.
       - Supports real-time layout changes (width, height, spacing, fonts) pushed directly from the UI.
       - Automatically manages lifecycle cleanups on character respawn and player removal.

    4. ANIMATION STATE OBSERVER & EVENT SYSTEM:
       - Hooks directly into character Animators to intercept play tracks.
       - Detects "Front Dash" and "Side Dash" using accurate animation IDs.
       - Handles the local player and remote players differently, minimizing network and CPU load.
       - "Evasive" detection tracks Workspace descendants to intercept the "RagdollCancel" trigger immediately.
--]]

-- ── MATH & INTERPOLATION UTILITIES ──────────────────────────────────────────────────────────────
----------------------------------------------------------------------------------------------------
--[[
    - Standard Time-Based Exponential Decay (Lerp):
      Allows smooth UI movement independent of frame rate. Instead of standard linear interpolation, 
      we use an exponential decay function:
      Value = Target + (Value - Target) * math.exp(-Speed * dt)
--]]

local MathUtils = {}

function MathUtils.lerp(a, b, t)
    return a + (b - a) * math.clamp(t, 0, 1)
end

function MathUtils.colorLerp(c1, c2, t)
    return Color3.new(
        MathUtils.lerp(c1.R, c2.R, t),
        MathUtils.lerp(c1.G, c2.G, t),
        MathUtils.lerp(c1.B, c2.B, t)
    )
end

function MathUtils.randomVector2(minX, maxX, minY, maxY)
    return Vector2.new(
        math.random() * (maxX - minX) + minX,
        math.random() * (maxY - minY) + minY
    )
end

-- ── CUSTOM UI PHYSICS PARTICLE ENGINE (LumeParticleEngine) ───────────────────────────────────────
----------------------------------------------------------------------------------------------------
--[[
    This engine leverages an ObjectPool pattern. Performance is critical because standard UI scripts
    create and destroy frame objects constantly, which triggers severe GC collection pauses.
    With LumeParticleEngine, we instantiate a pool of Frames and UIStroke elements, simply recycling
    them dynamically as they live and die.
--]]

local LumeParticleEngine = {}
LumeParticleEngine.Particles = {}
LumeParticleEngine.Pool = {}
LumeParticleEngine.ActiveCount = 0

function LumeParticleEngine.GetFromPool(parent)
    local pool = LumeParticleEngine.Pool
    if #pool > 0 then
        local pFrame = table.remove(pool)
        pFrame.Parent = parent
        pFrame.Visible = true
        return pFrame
    else
        local pFrame = Instance.new("Frame")
        pFrame.BorderSizePixel = 0
        pFrame.Name = "LumeSpark"
        pFrame.ZIndex = 8
        
        local corner = Instance.new("UICorner", pFrame)
        corner.CornerRadius = UDim.new(1, 0)
        
        local stroke = Instance.new("UIStroke", pFrame)
        stroke.Thickness = 0.8
        stroke.Color = WHITE
        stroke.Transparency = 0.5
        
        return pFrame
    end
end

function LumeParticleEngine.ReturnToPool(pFrame)
    pFrame.Visible = false
    pFrame.Parent = nil
    table.insert(LumeParticleEngine.Pool, pFrame)
end

function LumeParticleEngine.Spawn(parent, worldPos, velocity, startColor, endColor)
    if not Toggles.ParticlesOn.Value then return end
    if LumeParticleEngine.ActiveCount >= Options.PartMaxActive.Value then return end
    
    local pFrame = LumeParticleEngine.GetFromPool(parent)
    
    local startSize  = Options.PartStartSize.Value
    local endSize    = Options.PartEndSize.Value
    local startAlpha = Options.PartStartAlpha.Value
    local endAlpha   = Options.PartEndAlpha.Value
    local maxLife    = Options.PartLifetime.Value
    
    pFrame.Size = UDim2.new(0, startSize, 0, startSize)
    pFrame.Position = UDim2.new(0, worldPos.X, 0, worldPos.Y)
    pFrame.BackgroundColor3 = startColor
    pFrame.BackgroundTransparency = 1 - startAlpha
    
    local stroke = pFrame:FindFirstChildOfClass("UIStroke")
    if stroke then
        stroke.Transparency = (1 - startAlpha) + 0.15
        stroke.Color = startColor
    end
    
    local particle = {
        frame       = pFrame,
        stroke      = stroke,
        position    = worldPos,
        velocity    = velocity,
        startColor  = startColor,
        endColor    = endColor,
        startSize   = startSize,
        endSize     = endSize,
        startAlpha  = startAlpha,
        endAlpha    = endAlpha,
        lifetime    = maxLife,
        elapsed     = 0,
        rainbow     = Toggles.PartRainbow.Value
    }
    
    table.insert(LumeParticleEngine.Particles, particle)
    LumeParticleEngine.ActiveCount = LumeParticleEngine.ActiveCount + 1
end

function LumeParticleEngine.TriggerBurst(parent, centerPos, quantity, startColor, endColor)
    if not Toggles.ParticlesOn.Value or not Toggles.ParticleExplosion.Value then return end
    for i = 1, quantity do
        local angle = math.random() * math.pi * 2
        local speed = math.random(60, 280)
        local vel = Vector2.new(math.cos(angle) * speed, math.sin(angle) * speed)
        LumeParticleEngine.Spawn(parent, centerPos, vel, startColor, endColor)
    end
end

function LumeParticleEngine.Update(dt)
    local gravity = Options.PartGravity.Value
    local wind    = Options.PartWind.Value
    local drag    = Options.PartDrag.Value
    local particles = LumeParticleEngine.Particles
    local active = LumeParticleEngine.ActiveCount
    
    for i = #particles, 1, -1 do
        local p = particles[i]
        p.elapsed = p.elapsed + dt
        
        if p.elapsed >= p.lifetime then
            LumeParticleEngine.ReturnToPool(p.frame)
            table.remove(particles, i)
            LumeParticleEngine.ActiveCount = LumeParticleEngine.ActiveCount - 1
        else
            -- Physics calculation: Euler integration with friction drag force
            local speed = p.velocity.Magnitude
            local dragForce = Vector2.new(0, 0)
            if speed > 0.01 then
                dragForce = p.velocity.Unit * (speed * drag * dt)
            end
            
            local externalForces = Vector2.new(wind * dt, gravity * dt)
            p.velocity = p.velocity + externalForces - dragForce
            p.position = p.position + p.velocity * dt
            
            -- Dynamic styling calculation
            local progress = p.elapsed / p.lifetime
            local size = MathUtils.lerp(p.startSize, p.endSize, progress)
            local alpha = MathUtils.lerp(p.startAlpha, p.endAlpha, progress)
            
            local color = p.startColor
            if p.rainbow then
                color = Color3.fromHSV((progress + tick() * 0.15) % 1, 0.9, 0.95)
            else
                color = MathUtils.colorLerp(p.startColor, p.endColor, progress)
            end
            
            p.frame.Size = UDim2.new(0, size, 0, size)
            p.frame.Position = UDim2.new(0, p.position.X, 0, p.position.Y)
            p.frame.BackgroundColor3 = color
            p.frame.BackgroundTransparency = 1 - alpha
            
            if p.stroke then
                p.stroke.Transparency = (1 - alpha) + 0.15
                p.stroke.Color = color
            end
        end
    end
end

-- Hook the physical loop into Roblox Frame tick
RunService.Heartbeat:Connect(function(dt)
    LumeParticleEngine.Update(dt)
end)

function ThemeManager.Apply(themeName)
    local style = ThemeManager.Themes[themeName]
    if not style then return end
    
    for key, color in pairs(style) do
        local picker = Options[key]
        if picker then
            picker:SetValueRGB(color)
        end
    end
end

local function startEnchantSweep(sweep, delay)
    task.spawn(function()
        task.wait(delay or 0)
        while sweep and sweep.Parent do
            pcall(function()
                sweep.Position = UDim2.new(-0.45, 0, -0.1, 0)
            end)
            local t = TweenService:Create(sweep,
                TweenInfo.new(2.4, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut),
                { Position = UDim2.new(1.15, 0, -0.1, 0) }
            )
            t:Play()
            task.wait(3.2)
        end
    end)
end

local function startSparkle(sp, delay)
    task.spawn(function()
        task.wait(delay or 0)
        while sp and sp.Parent do
            task.wait(math.random(12, 36) * 0.1)
            if not (sp and sp.Parent) then break end
            sp.BackgroundTransparency = 0.05
            task.wait(0.06)
            sp.BackgroundTransparency = 0.45
            task.wait(0.06)
            sp.BackgroundTransparency = 0.05
            TweenService:Create(sp,
                TweenInfo.new(0.7, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                { BackgroundTransparency = 1 }
            ):Play()
        end
    end)
end

local function doPulse(barData)
    if not Toggles.PulseOn.Value then return end
    if barData.pulseActive then return end
    barData.pulseActive = true
    
    local frame   = barData.frame
    local scale   = Options.PulseSc.Value or 1.07
    local halfDur = (Options.PulseDur.Value or 0.38) * 0.5
    local orig    = UDim2.new(0, Options.Width.Value, 0, Options.Height.Value)
    local scaled  = UDim2.new(
        0, Options.Width.Value * scale,
        0, Options.Height.Value * scale
    )
    
    -- Gentle offset positioning to maintain central alignment
    local xShift = -(Options.Width.Value * (scale - 1)) / 2
    local yShift = -(Options.Height.Value * (scale - 1)) / 2
    local lblH   = Toggles.ShowLabels.Value and (Options.LabelSize.Value + 3) or 0
    local basePos = UDim2.new(0, 0, 0, lblH + (Toggles.ShowLabels.Value and 2 or 0))
    local shiftedPos = UDim2.new(0, xShift, 0, lblH + yShift)
    
    TweenService:Create(frame,
        TweenInfo.new(halfDur, Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
        { Size = scaled, Position = shiftedPos }
    ):Play()
    
    task.delay(halfDur, function()
        TweenService:Create(frame,
            TweenInfo.new(halfDur, Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
            { Size = orig, Position = basePos }
        ):Play()
        task.delay(halfDur + 0.05, function()
            barData.pulseActive = false
        end)
    end)
end

local function doReadyFlash(barData)
    if not Toggles.FlashOn.Value then return end
    if barData.readyFlashing then return end
    barData.readyFlashing = true
    
    local fill     = barData.fill
    local flashClr = Options.FlashClr and Options.FlashClr.Value or Color3.fromRGB(255, 255, 100)
    local origClr  = fill.BackgroundColor3
    local count    = Options.FlashCnt.Value or 3
    local spd      = Options.FlashSpd.Value or 0.14
    
    local function flashN(n)
        if n <= 0 then
            pcall(function() fill.BackgroundColor3 = origClr end)
            barData.readyFlashing = false
            return
        end
        TweenService:Create(fill,
            TweenInfo.new(spd, Enum.EasingStyle.Linear),
            { BackgroundColor3 = flashClr }
        ):Play()
        task.delay(spd, function()
            TweenService:Create(fill,
                TweenInfo.new(spd, Enum.EasingStyle.Linear),
                { BackgroundColor3 = origClr }
            ):Play()
            task.delay(spd, function() flashN(n - 1) end)
        end)
    end
    
    task.spawn(flashN, count)
end

local function doReadyPulse(barData)
    if barData.readyPulseActive then return end
    barData.readyPulseActive = true
    
    local stroke    = barData.barStroke
    local glow      = barData.glow
    local outerGlow = barData.outerGlow
    
    local function pulse()
        if not barData.readyPulseActive then return end
        -- Breathe in (neon intensify)
        TweenService:Create(stroke, TweenInfo.new(1.0, Enum.EasingStyle.Sine), { Transparency = 0.0 }):Play()
        TweenService:Create(glow, TweenInfo.new(1.0, Enum.EasingStyle.Sine), { BackgroundTransparency = 0.48 }):Play()
        TweenService:Create(outerGlow, TweenInfo.new(1.0, Enum.EasingStyle.Sine), { BackgroundTransparency = 0.76 }):Play()
        task.wait(1.0)
        
        if not barData.readyPulseActive then return end
        -- Breathe out (neon dim)
        TweenService:Create(stroke, TweenInfo.new(1.0, Enum.EasingStyle.Sine), { Transparency = 0.58 }):Play()
        TweenService:Create(glow, TweenInfo.new(1.0, Enum.EasingStyle.Sine), { BackgroundTransparency = 0.84 }):Play()
        TweenService:Create(outerGlow, TweenInfo.new(1.0, Enum.EasingStyle.Sine), { BackgroundTransparency = 0.95 }):Play()
        task.wait(1.0)
        
        if barData.readyPulseActive then
            task.spawn(pulse)
        end
    end
    task.spawn(pulse)
end

local function stopReadyPulse(barData)
    barData.readyPulseActive = false
end

    local spacing    = Options.Spacing.Value
    local cornerR    = Options.Corner.Value
    local posX       = Options.PosX.Value
    local posY       = Options.PosY.Value
    local bgTransp   = Options.BGTransp.Value
    local fillTransp = Options.FillTransp.Value
    local textSize   = Options.TextSize.Value
    local lblSize    = Options.LabelSize.Value
    local lerpSpd    = Options.FillSpeed.Value
    local fillDir    = Options.FillDir.Value
    local font       = FONT_MAP[Options.FontChoice.Value] or Enum.Font.GothamBold
    
    local showBars   = Toggles.ShowBars.Value
    local showNums   = Toggles.ShowNumbers.Value
    local showLabels = Toggles.ShowLabels.Value
    local showReady  = Toggles.ShowReady.Value
    local showPct    = Toggles.ShowPercent.Value
    local showBorder = Toggles.ShowBorder.Value
    local showShine  = Toggles.ShowShine.Value
    local showGlow   = Toggles.ShowGlow.Value
    
    local totalW = #ABILITY_ORDER * width + (#ABILITY_ORDER - 1) * spacing
    local startX = -totalW / 2
    
    for i, name in ipairs(ABILITY_ORDER) do
        local data = bars[name]
        local cfg  = ABILITY_CFG[name]
        
        if data.time > 0 then
            data.time = math.max(data.time - dt, 0)
        end
        
        local realRatio = 1 - (data.time / math.max(data.duration, 0.001))
        local lf        = math.min(lerpSpd * dt, 1)
        data.visRatio   = data.visRatio + (realRatio - data.visRatio) * lf
        
        local isOnCd = data.time > 0
        
        -- Transitions when state changes
        if data.prevOnCd and not isOnCd then
            task.spawn(doReadyFlash, data)
            task.spawn(doReadyPulse, data)
            
            -- Explosion particles on Cooldown Ready!
            if Toggles.ParticlesOn.Value and Toggles.ParticleExplosion.Value then
                local frameAbsPos = data.frame.AbsolutePosition
                local frameAbsSize = data.frame.AbsoluteSize
                local center = frameAbsPos + Vector2.new(frameAbsSize.X / 2, frameAbsSize.Y / 2)
                local sparkClr = Options[cfg.fillKey].Value
                LumeParticleEngine.TriggerBurst(data.frame, center, 22, sparkClr, WHITE)
            end
        elseif not data.prevOnCd and isOnCd then
            stopReadyPulse(data)
        end
        data.prevOnCd = isOnCd
        
        -- Emitter stream along the leading edge of the progress fill
        if isOnCd and Toggles.ParticlesOn.Value then
            local now = tick()
            local spawnRate = Options.PartSpawnRate.Value
            if now - data.lastPartSpawn >= spawnRate then
                data.lastPartSpawn = now
                
                local frameAbsPos = data.frame.AbsolutePosition
                local frameAbsSize = data.frame.AbsoluteSize
                
                local currentFillWidth = frameAbsSize.X * math.clamp(data.visRatio, 0, 1)
                local leadingX = 0
                if fillDir == "Left" then
                    leadingX = currentFillWidth
                else
                    leadingX = frameAbsSize.X - currentFillWidth
                end
                
                local spawnPos = Vector2.new(leadingX, math.random(2, frameAbsSize.Y - 2))
                local sparkClr = Options[cfg.fillKey].Value
                local speedY = math.random(-40, 40)
                local speedX = (fillDir == "Left" and -1 or 1) * math.random(15, 60)
                
                LumeParticleEngine.Spawn(data.fill, spawnPos, Vector2.new(speedX, speedY), sparkClr, WHITE)
            end
        end
        
        local xOff = startX + (i - 1) * (width + spacing)
        local lblH  = showLabels and (lblSize + 3) or 0
        local totH  = height + lblH + (showLabels and 2 or 0)
        
        data.container.Size     = UDim2.new(0, width, 0, totH)
        data.container.Position = UDim2.new(posX, xOff, posY, -totH / 2)
        
        -- Title configuration
        data.abilityLabel.Size       = UDim2.new(1, 0, 0, lblH)
        data.abilityLabel.Position   = UDim2.new(0, 2, 0, 0)
        data.abilityLabel.TextSize   = lblSize
        data.abilityLabel.Font       = font
        data.abilityLabel.TextColor3 = Options[cfg.lblKey].Value
        data.abilityLabel.Visible    = showLabels
        
        local fillClr   = Options[cfg.fillKey].Value
        local bgClr     = Options[cfg.bgKey].Value
        local numClr    = Options[cfg.numKey].Value
        local borderClr = Options[cfg.borderKey].Value
        
        local ratio = math.clamp(data.visRatio, 0, 1)
        local hasFill = ratio > 0.03
        
        -- Dynamic sizing for Glow Elements
        data.outerGlow.Size             = UDim2.new(0, width + 26, 0, height + 18)
        data.outerGlow.Position         = UDim2.new(0, -13, 0, lblH - 7)
        data.outerGlow.BackgroundColor3 = fillClr
        data.outerGlow.Visible          = showGlow and hasFill and not data.readyPulseActive
        
        data.glow.Size             = UDim2.new(0, width + 10, 0, height + 8)
        data.glow.Position         = UDim2.new(0, -5, 0, lblH - 3)
        data.glow.BackgroundColor3 = fillClr
        data.glow.Visible          = showGlow and hasFill and not data.readyPulseActive
        
        -- Sparkling parameters
        for _, sp in ipairs(data.sparkles) do
            sp.BackgroundColor3 = fillClr
        end
        
        if not data.pulseActive then
            data.frame.Size     = UDim2.new(0, width, 0, height)
            data.frame.Position = UDim2.new(0, 0, 0, lblH + (showLabels and 2 or 0))
        end
        
        data.barCorner.CornerRadius  = UDim.new(0, cornerR)
        data.fillCorner.CornerRadius = UDim.new(0, cornerR)
        
        data.barStroke.Enabled = showBorder
        data.barStroke.Color   = borderClr
        if not data.readyPulseActive then
            data.barStroke.Transparency = 0.15
        end
        
        data.shine.Visible     = showShine
        data.shineLine.Visible = showShine
        
        if showBars then
            data.container.Visible            = true
            data.frame.Visible                = true
            data.fill.Visible                 = true
            data.frame.BackgroundColor3       = bgClr
            data.frame.BackgroundTransparency = bgTransp
            data.fill.BackgroundColor3        = fillClr
            data.fill.BackgroundTransparency  = fillTransp
            
            if fillDir == "Left" then
                data.fill.AnchorPoint = Vector2.new(0, 0)
                data.fill.Position    = UDim2.new(0, 0, 0, 0)
            else
                data.fill.AnchorPoint = Vector2.new(1, 0)
                data.fill.Position    = UDim2.new(1, 0, 0, 0)
            end
            data.fill.Size = UDim2.new(ratio, 0, 1, 0)
            
        elseif showNums then
            data.container.Visible            = true
            data.frame.Visible                = true
            data.fill.Visible                 = false
            data.frame.BackgroundTransparency = 1
        else
            data.container.Visible = false
        end
        
        if data.container.Visible then
            data.text.Font     = font
            data.text.TextSize = textSize
            if showNums then
                if isOnCd then
                    data.text.TextColor3 = numClr
                    data.text.Text = showPct
                        and string.format("%.0f%%", realRatio * 100)
                        or  string.format("%.1f",   data.time)
                elseif showReady then
                    data.text.Text       = "✦ READY ✦"
                    data.text.TextColor3 = ENCHANT_MINT
                else
                    data.text.Text = ""
                end
            else
                data.text.Text = ""
            end
        end
    end
end)

local ANIM_MAP = {
    { name = "Dash", ids = { 10479335397, 10491993682 } },

    local data = bars[name]
    if not data then return end
    data.time     = data.duration
    data.prevOnCd = true
    stopReadyPulse(data)
    task.spawn(doPulse, data)
end

local function onLocalAnimId(id)
    for _, entry in ipairs(ANIM_MAP) do
        for _, animId in ipairs(entry.ids) do
            if id == animId then
                triggerLocal(entry.name)
            end
        end
    end
end

    local char     = player.Character or player.CharacterAdded:Wait()
    local hum      = char:WaitForChild("Humanoid", 10)
    local animator = hum and hum:WaitForChild("Animator", 10)
    if not animator then return end
    
    animator.AnimationPlayed:Connect(function(track)
        local id = tonumber(track.Animation.AnimationId:match("%d+"))
        if id then
            onLocalAnimId(id)
        end
    end)
end

player.CharacterAdded:Connect(function()
    task.spawn(detectLocal)
end)

    task.spawn(detectLocal)
end

--[[
    This engine dynamically creates miniature structural progress bars directly above other players'
    characters. It provides customizable sizes, spacing, neon glow effects, and gradient fills matching
    the targeted Obsidian layout.
--]]

local OH_CFG = {
    Dash = {
        label = "FRONT DASH", defaultCD = 5,
        txtKey = "OHDashTxt", bgKey = "OHDashBG", fillKey = "DashFill"
    },
    Side = {
        label = "SIDE DASH", defaultCD = 2,
        txtKey = "OHSideTxt", bgKey = "OHSideBG", fillKey = "SideFill"
    },
    Evasive = {
        label = "EVASIVE", defaultCD = 30,
        txtKey = "OHEvaTxt", bgKey = "OHEvaBG", fillKey = "EvaFill"
    },
}
local OH_ORDER = { "Dash", "Side", "Evasive" }
local playerTrackers = {}

local function buildOverhead(p)
    local char = p.Character
    if not char then return end
    local head = char:FindFirstChild("Head")
    if not head then return end
    
    if playerTrackers[p] and playerTrackers[p]._billboard then
        pcall(function() playerTrackers[p]._billboard:Destroy() end)
    end
    
    local ohW    = Options.OHW.Value
    local ohH    = Options.OHH.Value
    local ohSp   = Options.OHSp.Value
    local corner = Options.OHCorner.Value
    local stroke = Options.OHStroke.Value
    
    local totH = #OH_ORDER * ohH + (#OH_ORDER - 1) * ohSp
    
    local bill = Instance.new("BillboardGui")
    bill.Name           = "CDOverhead"
    bill.Adornee        = head
    bill.Size           = UDim2.new(0, ohW, 0, totH)
    bill.StudsOffset    = Vector3.new(0, Options.OHStuds.Value, 0)
    bill.AlwaysOnTop    = Toggles.OHTop.Value
    bill.LightInfluence = 0
    bill.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    bill.Parent         = head
    
    local layout = Instance.new("UIListLayout", bill)
    layout.SortOrder           = Enum.SortOrder.LayoutOrder
    layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
    layout.FillDirection       = Enum.FillDirection.Vertical
    layout.Padding             = UDim.new(0, ohSp)
    
    local barRefs = {}
    
    for idx, abilityName in ipairs(OH_ORDER) do
        local cfg = OH_CFG[abilityName]
        
        local bg = Instance.new("Frame")
        bg.Name                   = abilityName .. "BarFrame"
        bg.Size                   = UDim2.new(1, 0, 0, ohH)
        bg.BackgroundColor3       = Options[cfg.bgKey].Value
        bg.BackgroundTransparency = Options.OHBGTransp.Value
        bg.BorderSizePixel        = 0
        bg.LayoutOrder            = idx
        bg.Visible                = false
        bg.Parent                 = bill
        
        local bgCorner = Instance.new("UICorner", bg)
        bgCorner.CornerRadius = UDim.new(0, corner)
        
        -- Glowing effects inside Billboard
        local ohGlow = Instance.new("Frame")
        ohGlow.Name = "OHGlow"
        ohGlow.Size = UDim2.new(1, 8, 1, 6)
        ohGlow.Position = UDim2.new(0, -4, 0, -3)
        ohGlow.ZIndex = 1
        ohGlow.BorderSizePixel = 0
        ohGlow.BackgroundTransparency = 0.8
        ohGlow.Visible = Toggles.OHGlowShow.Value
        ohGlow.Parent = bg
        Instance.new("UICorner", ohGlow).CornerRadius = UDim.new(0, corner + 2)
        
        -- Clip frame for opponent bars
        local ohClip = Instance.new("Frame")
        ohClip.Name = "OHClip"
        ohClip.Size = UDim2.new(1, 0, 1, 0)
        ohClip.BackgroundTransparency = 1
        ohClip.BorderSizePixel = 0
        ohClip.ClipsDescendants = true
        ohClip.ZIndex = 2
        ohClip.Parent = bg
        
        -- Glowing Fills
        local ohFill = Instance.new("Frame")
        ohFill.Name = "OHFill"
        ohFill.BorderSizePixel = 0
        ohFill.ZIndex = 2
        ohFill.Parent = ohClip
        local ohFillCorner = Instance.new("UICorner", ohFill)
        ohFillCorner.CornerRadius = UDim.new(0, corner)
        
        -- Opponent iridescent shimmer gradients
        local ohFillGrad = Instance.new("UIGradient")
        ohFillGrad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0,    WHITE),
            ColorSequenceKeypoint.new(0.40, Color3.fromRGB(190, 175, 255)),
            ColorSequenceKeypoint.new(1,    Color3.fromRGB(40, 25, 110)),
        })
        ohFillGrad.Rotation = 90
        ohFillGrad.Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.45),
            NumberSequenceKeypoint.new(1, 0.20),
        })
        ohFillGrad.Parent = ohFill
        
        -- Reflected top shine line
        local ohShine = Instance.new("Frame")
        ohShine.Name = "OHShine"
        ohShine.Size = UDim2.new(1, 0, 0.45, 0)
        ohShine.BackgroundTransparency = 0.85
        ohShine.BackgroundColor3 = WHITE
        ohShine.BorderSizePixel = 0
        ohShine.Visible = Toggles.OHShineShow.Value
        ohShine.ZIndex = 3
        ohShine.Parent = ohFill
        Instance.new("UICorner", ohShine)
        
        local bgStroke = Instance.new("UIStroke", bg)
        bgStroke.Color           = Options.OHBorder.Value
        bgStroke.Thickness       = stroke
        bgStroke.Transparency    = 0.30
        bgStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
        
        local ohFont = FONT_MAP[Options.OHFontChoice.Value] or Enum.Font.GothamBold
        
        -- Opponent Timer label overlays
        local lbl = Instance.new("TextLabel")
        lbl.Name                   = abilityName .. "Text"
        lbl.Size                   = UDim2.new(1, -8, 1, 0)
        lbl.Position               = UDim2.new(0, 4, 0, 0)
        lbl.BackgroundTransparency = 1
        lbl.Text                   = ""
        lbl.TextColor3             = Options[cfg.txtKey].Value
        lbl.TextScaled             = false
        lbl.TextSize               = ohH * 0.62
        lbl.Font                   = ohFont
        lbl.TextXAlignment         = Enum.TextXAlignment.Center
        lbl.TextYAlignment         = Enum.TextYAlignment.Center
        lbl.TextStrokeTransparency = Options.OHStroke.Value
        lbl.TextStrokeColor3       = BLACK
        lbl.ZIndex                 = 4
        lbl.Parent                 = bg
        
        barRefs[abilityName] = {
            bg     = bg,
            glow   = ohGlow,
            fill   = ohFill,
            shine  = ohShine,
            lbl    = lbl,
            stroke = bgStroke,
            grad   = ohFillGrad,
            lastPartSpawn = 0
        }
    end
    
    if not playerTrackers[p] then
        playerTrackers[p] = {}
    end
    playerTrackers[p]._billboard = bill
    
    for abilityName in pairs(OH_CFG) do
        if not playerTrackers[p][abilityName] then
            playerTrackers[p][abilityName] = { time = 0, visRatio = 0 }
        end
        local refs = barRefs[abilityName]
        if refs then
            playerTrackers[p][abilityName].bg            = refs.bg
            playerTrackers[p][abilityName].glow          = refs.glow
            playerTrackers[p][abilityName].fill          = refs.fill
            playerTrackers[p][abilityName].shine         = refs.shine
            playerTrackers[p][abilityName].label         = refs.lbl
            playerTrackers[p][abilityName].stroke        = refs.stroke
            playerTrackers[p][abilityName].grad          = refs.grad
            playerTrackers[p][abilityName].lastPartSpawn = 0
        end
    end
end

local function triggerOverhead(p, abilityName)
    local cfg = OH_CFG[abilityName]
    if not cfg then return end
    
    if not playerTrackers[p] then
        playerTrackers[p] = {}
    end
    if not playerTrackers[p][abilityName] then
        playerTrackers[p][abilityName] = { time = 0, visRatio = 0 }
    end
    
    local tracker = playerTrackers[p][abilityName]
    tracker.time = cfg.defaultCD
    
    -- Animate opponent overhead triggers
    local bg = tracker.bg
    if bg then
        local origSize = UDim2.new(1, 0, 0, Options.OHH.Value)
        local scaled = UDim2.new(1.08, 0, 0, Options.OHH.Value * 1.15)
        TweenService:Create(bg, TweenInfo.new(0.18, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), { Size = scaled }):Play()
        task.delay(0.18, function()
            TweenService:Create(bg, TweenInfo.new(0.18, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), { Size = origSize }):Play()
        end)
    end
end

local function setupPlayer(p)
    if p == player then return end
    
    playerTrackers[p] = {}
    for abilityName in pairs(OH_CFG) do
        playerTrackers[p][abilityName] = { time = 0, visRatio = 0 }
    end
    
    local function onChar(char)
        char:WaitForChild("Head", 10)
        task.wait(0.12)
        buildOverhead(p)
        
        local hum  = char:WaitForChild("Humanoid", 10)
        local anim = hum and hum:WaitForChild("Animator", 10)
        if not anim then return end
        
        anim.AnimationPlayed:Connect(function(track)
            local id = tonumber(track.Animation.AnimationId:match("%d+"))
            if not id then return end
            
            for _, entry in ipairs(ANIM_MAP) do
                for _, animId in ipairs(entry.ids) do
                    if id == animId then
                        triggerOverhead(p, entry.name)
                    end
                end
            end
        end)
    end
    
    if p.Character then
        task.spawn(onChar, p.Character)
    end
    p.CharacterAdded:Connect(onChar)
end

for _, p in ipairs(Players:GetPlayers()) do
    setupPlayer(p)
end
Players.PlayerAdded:Connect(setupPlayer)

Players.PlayerRemoving:Connect(function(p)
    local tracker = playerTrackers[p]
    if not tracker then return end
    
    if tracker._billboard then
        pcall(function() tracker._billboard:Destroy() end)
    end
    playerTrackers[p] = nil
end)

-- ── OVERHEAD BILLBOARDS HEARTBEAT RENDERING LOOP ─────────────────────────────────────────────────
----------------------------------------------------------------------------------------------------
RunService.Heartbeat:Connect(function(dt)
    local showOH   = Toggles.ShowOH.Value
    local showBG   = Toggles.OHBGShow.Value
    local onlyCd   = Toggles.OHOnlyCd.Value
    local showBord = Toggles.OHBorderShow.Value
    local showGlow = Toggles.OHGlowShow.Value
    local showShin = Toggles.OHShineShow.Value
    
    local ohW      = Options.OHW.Value
    local ohH      = Options.OHH.Value
    local ohSp     = Options.OHSp.Value
    local ohSO     = Options.OHStuds.Value
    local ohBGT    = Options.OHBGTransp.Value
    local ohStr    = Options.OHStroke.Value
    local ohAlTop  = Toggles.OHTop.Value
    local borderC  = Options.OHBorder.Value
    local corner   = Options.OHCorner.Value
    local stroke   = Options.OHStroke.Value
    local fillDir  = Options.FillDir.Value
    local lerpSpd  = Options.FillSpeed.Value
    
    local totH     = #OH_ORDER * ohH + (#OH_ORDER - 1) * ohSp
    local ohFont   = FONT_MAP[Options.OHFontChoice.Value] or Enum.Font.GothamBold
    
    for p, tracker in pairs(playerTrackers) do
        local bill = tracker._billboard
        if bill then
            bill.Enabled     = showOH
            bill.AlwaysOnTop = ohAlTop
            bill.StudsOffset = Vector3.new(0, ohSO, 0)
            bill.Size        = UDim2.new(0, ohW, 0, totH)
        end
        
        for abilityName, data in pairs(tracker) do
            if abilityName == "_billboard" then continue end
            if data.time > 0 then
                data.time = math.max(data.time - dt, 0)
            end
            
            local bg   = data.bg
            local glow = data.glow
            local fill = data.fill
            local shin = data.shine
            local lbl  = data.label
            local str  = data.stroke
            local cfg  = OH_CFG[abilityName]
            
            if not cfg or not lbl or not bg then continue end
            if not lbl.Parent or not bg.Parent then continue end
            
            -- Opponent exponential fill decays
            local realRatio = 1 - (data.time / math.max(cfg.defaultCD, 0.001))
            local lf        = math.min(lerpSpd * dt, 1)
            data.visRatio   = data.visRatio + (realRatio - data.visRatio) * lf
            local ratio     = math.clamp(data.visRatio, 0, 1)
            
            -- Opponent leading spark emitters
            if data.time > 0 and Toggles.ParticlesOn.Value then
                local now = tick()
                if now - data.lastPartSpawn >= Options.PartSpawnRate.Value * 1.5 then
                    data.lastPartSpawn = now
                    
                    local sparkClr = Options[cfg.fillKey].Value
                    local bgAbsSize = bg.AbsoluteSize
                    
                    if bgAbsSize.X > 5 then
                        local leadingX = ratio * bgAbsSize.X
                        if fillDir == "Right" then
                            leadingX = bgAbsSize.X - leadingX
                        end
                        local spawnPos = Vector2.new(leadingX, math.random(2, bgAbsSize.Y - 2))
                        local speedY = math.random(-25, 25)
                        local speedX = (fillDir == "Left" and -1 or 1) * math.random(10, 40)
                        
                        LumeParticleEngine.Spawn(fill, spawnPos, Vector2.new(speedX, speedY), sparkClr, WHITE)
                    end
                end
            end
            
            -- Style parameter updates
            lbl.TextColor3             = Options[cfg.txtKey].Value
            lbl.TextStrokeTransparency = ohStr
            lbl.TextSize               = ohH * 0.62
            lbl.Font                   = ohFont
            
            if showBG then
                bg.BackgroundTransparency = ohBGT
                bg.BackgroundColor3       = Options[cfg.bgKey].Value
            else
                bg.BackgroundTransparency = 1
            end
            
            if glow then
                glow.BackgroundColor3 = Options[cfg.fillKey].Value
                glow.Visible = showGlow and (ratio > 0.03) and (data.time > 0)
                local glowCorner = glow:FindFirstChildOfClass("UICorner")
                if glowCorner then glowCorner.CornerRadius = UDim.new(0, corner + 2) end
            end
            
            if fill then
                fill.BackgroundColor3 = Options[cfg.fillKey].Value
                fill.Visible = showOH and (data.time > 0)
                
                if fillDir == "Left" then
                    fill.AnchorPoint = Vector2.new(0, 0)
                    fill.Position    = UDim2.new(0, 0, 0, 0)
                else
                    fill.AnchorPoint = Vector2.new(1, 0)
                    fill.Position    = UDim2.new(1, 0, 0, 0)
                end
                fill.Size = UDim2.new(ratio, 0, 1, 0)
                
                local fillCorner = fill:FindFirstChildOfClass("UICorner")
                if fillCorner then fillCorner.CornerRadius = UDim.new(0, corner) end
            end
            
            if shin then
                shin.Visible = showShin
            end
            
            if str then
                str.Enabled      = showBord
                str.Color        = borderC
                str.Thickness    = stroke
                str.Transparency = 0.30
            end
            
            local bgCorner = bg:FindFirstChildOfClass("UICorner")

    1. THE PHYSICS OF LUME PARTICLES
       The particle simulation utilizes standard Newtonian kinematics computed in a two-dimensional
       coordinate space (representing the pixel-grid of the Roblox ScreenGui / BillboardGui).
       
       Mathematical Formulas:
       Let P_t = Position at frame t
       Let V_t = Velocity at frame t
       Let A_t = Acceleration at frame t
       
       We apply Euler Integration:
       V_t = V_{t-1} + (A_t * dt)
       P_t = P_{t-1} + (V_t * dt)
       
       To simulate fluid drag or air resistance, we define a friction coefficients (F_drag) that
       acts in the exact opposite direction of the particle velocity vector:
       F_drag = V_t.Unit * (V_t.Magnitude^2 * FrictionFactor)
       
       Applying Drag:
       V_t = V_t - (F_drag * dt)
       
       By recalculating these equations during every tick of the Heartbeat service (which triggers
       immediately before rendering), we provide a smooth motion pattern without lag.
       
    2. THE MEMORY CONSTRAINTS OF ROBLOX GUI INSTANCES
       Roblox is notorious for garbage collection stutters when scripts instantiate many frames 
       simultaneously and immediately delete them. Debris:AddItem() or Instance:Destroy() forces
       the Lua engine to search for orphans and perform sweeping, causing CPU spikes.
       
       To prevent this:
       We built the LumeParticleEngine Pool.
       When a particle is spawned:
         - The engine first checks if LumeParticleEngine.Pool contains any retired frames.
         - If a frame is available, it is popped, parented, and configured.
         - If the pool is empty, a new Frame is created and formatted.
       When a particle's lifetime expires:
         - The frame is unparented, set to Invisible, and pushed back into the Pool array.
         - The object reference remains alive in memory, completely bypassing garbage collection.
         
    3. THE GRADIENT TRANSITIONS & SHIMMER MATH
       Roblox UIGradients utilize a ColorSequence made of ColorSequenceKeypoints. Dynamic color 
       transition is achieved by updating these keypoints in real time.

       TweenInfo: EasingStyle.Quad, EasingDirection.InOut

    ================================================================================================
                               ADVANCED MATHEMATICAL & PHYSICAL PROOFS
    ================================================================================================
    To provide a deeper foundation for the physics and graphics methods used in this cooldown tracker, 
    below are detailed mathematical derivations for key modules:

    A. EXPONENTIAL DECAY DERIVATION (SMOOTH PROGRESS BAR FILL)
       Standard linear lerp updates the progress bar according to a linear step size, which causes 
       jerky movements if the frame rate fluctuates. Instead, we solve the following continuous 
       first-order differential equation:
       
       dy/dt = -k * (y - y_{target})
       
       Where:
         - y is the current visual ratio displayed on the screen.
         - y_{target} is the actual physical cooldown ratio (1 - time/duration).
         - k is the decay speed constant (controlled by Options.FillSpeed).
         
       Separating variables and integrating from t_0 to t:
       
       dy / (y - y_{target}) = -k * dt
       ln(y(t) - y_{target}) - ln(y(0) - y_{target}) = -k * (t - 0)
       ln( (y(t) - y_{target}) / (y(0) - y_{target}) ) = -k * t
       
       Exponentiating both sides:
       
       y(t) - y_{target} = (y(0) - y_{target}) * e^{-k * t}
       y(t) = y_{target} + (y(0) - y_{target}) * e^{-k * t}
       
       In our discrete codebase, dt is the DeltaTime passed by RunService.Heartbeat. Thus, the 
       recursive update equation applied every frame becomes:
       
       y_{new} = y_{target} + (y_{old} - y_{target}) * math.exp(-k * dt)
       
       This ensures absolute frame-rate independence: whether running at 30 FPS or 240 FPS, the visual 
       cooldown bar reaches the exact same position in the same physical time, without stuttering.

    B. FLUID DRAG FORCE AND PARTICLE VELOCITY
       Sparks emitted from the progress fill edge are subjected to two main forces: a constant 
       gravitational drift (G) and a fluid drag friction force (F_d).
       Roblox UI rendering operates in a 2D pixel space, so we represent the physics vectors as 
       Vector2 instances.
       
       Stokes' drag law states that the friction force is directly proportional to the velocity 
       magnitude at low velocities:
       
       F_drag = -c * V
       
       However, for premium glowing visual sparks, we implement a quadratic drag equation, which 
       accurately models turbulent airflow behind rapid particle bursts:
       
       F_drag = -0.5 * rho * C_d * A * |V|^2 * V.Unit
       
       We condense the constants (0.5 * rho * C_d * A) into a single user-adjustable friction parameter 
       "Drag" (represented as `drag` in our update loop). The discrete acceleration applied to the 
       particle during frame interval dt is:
       
       Acceleration = GravityVector + WindVector - (Drag * |V| * V)
       
       V_{new} = V_{old} + Acceleration * dt
       Position_{new} = Position_{old} + V_{new} * dt
       
       By using Euler Integration at 60Hz, the sparks trace natural fluid arcs, slowing down as they 
       spread out and float away.

    ================================================================================================
                             ANIMATION ASSET PRELOADING METHODOLOGY
    ================================================================================================
    To ensure that animation play detections trigger instantly without delay on first load, we recommend 
    utilizing Roblox's ContentProvider service. This pre-caches the animation assets in the client's local 
    device storage:

    ```lua
    local ContentProvider = game:GetService("ContentProvider")
    
    local function preloadAnimations()
        local assetsToLoad = {}
        for _, entry in ipairs(ANIM_MAP) do
            for _, id in ipairs(entry.ids) do
                local anim = Instance.new("Animation")
                anim.AnimationId = "rbxassetid://" .. tostring(id)
                table.insert(assetsToLoad, anim)
            end
        end
        
        -- Run the asynchronous preloader
        local success, err = pcall(function()
            ContentProvider:PreloadAsync(assetsToLoad)
        end)
        
        if success then
            print("[COOLDOWN TRACKER] Successfully preloaded all animations into cache!")
        else
            warn("[COOLDOWN TRACKER] Failed to preload animations: " .. tostring(err))
        end
    end
    
    task.spawn(preloadAnimations)
    ```

    Using `ContentProvider:PreloadAsync` guarantees that when another player executes a Side Dash or Front 
    Dash for the very first time in a gameplay match, the animation track fires the AnimationPlayed event 
    without any micro-freezes or initial delay spikes.
