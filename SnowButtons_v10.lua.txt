local function rnd(a, b)      return a + math.random() * (b - a) end
local function lerp(a, b, t)  return a + (b - a) * t             end
local function clamp(v,mn,mx) return math.max(mn, math.min(mx, v)) end
local function tw(obj, info, props) TweenSvc:Create(obj, info, props):Play() end
local function delay(t, fn)   task.delay(t, fn) end

local function newCorner(parent, r)
    local c = Instance.new("UICorner", parent)
    c.CornerRadius = UDim.new(0, r)
    return c
end
local function newStroke(parent, thick, col, trans)
    local s = Instance.new("UIStroke", parent)
    s.Thickness    = thick or 2
    s.Color        = col   or ICE.STROKE_IDLE
    s.Transparency = trans or 0
    return s
end
local function newGrad(parent, c1, c2, rot)
    local g = Instance.new("UIGradient", parent)
    g.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, c1),
        ColorSequenceKeypoint.new(1, c2),
    }
    g.Rotation = rot or 90
    return g
end
local function newGrad3(parent, c1, c2, c3, rot)
    local g = Instance.new("UIGradient", parent)
    g.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0,   c1),
        ColorSequenceKeypoint.new(0.5, c2),
        ColorSequenceKeypoint.new(1,   c3),
    }
    g.Rotation = rot or 90
    return g
end

-- ═══════════════════════════════════════════════════════════════
--  ДВИЖОК СНЕГОПАДА (мелкие снежинки, только внутри кнопки)
-- ═══════════════════════════════════════════════════════════════
local snowCharsAll = { "·","•","❄","❅","*","❆","·","•","·","*","❅","·" }

task.spawn(function()
    local time = 0
    while gui.Parent do
        local dt = task.wait(0.033)
        time = time + dt
        for _, sf in ipairs(AIM.snowflakes) do
            local sl = sf.label
            local py = sl.Position.Y.Scale + sf.speedY * dt * 30
            local wobX = math.sin(time * sf.wSpeed + sf.wobble) * 0.015
            local px = sf.startX + wobX
            sf.rot = (sf.rot + sf.rotSpd * dt) % 360
            if py > 1.14 then
                py = rnd(-0.18, 0.0)
                px = rnd(0, 0.90)
                sf.startX = px
                sl.Text     = snowCharsAll[math.random(#snowCharsAll)]
                sl.TextSize = math.random(2, 5)
                sl.TextTransparency = rnd(0.35, 0.75)
            end
            sl.Position = UDim2.new(clamp(px, 0, 0.92), 0, py, 0)
            sl.Rotation = sf.rot
        end
    end
end)

-- ═══════════════════════════════════════════════════════════════
--  ДВИЖОК КРИСТАЛЛОВ (мерцание угловых осколков)
-- ═══════════════════════════════════════════════════════════════
task.spawn(function()
    local time = 0
    while gui.Parent do
        time = time + task.wait(0.04)
        for i, cd in ipairs(AIM.crystals) do
            local phase = time * 0.62 + i * 1.1
            local s     = 0.5 + 0.5 * math.sin(phase)
            cd.label.TextTransparency = lerp(cd.baseTrans, cd.baseTrans - 0.32, s)
            cd.label.Rotation         = cd.baseRot + math.sin(phase * 0.8) * 13
            local scale = lerp(0.88, 1.0, s)
            cd.label.TextSize = math.max(4, math.floor(cd.baseSize * scale + 0.5))
        end
    end
end)

local function spawnRipple(parent, x, y, color)
    local ripple = Instance.new("Frame")
    ripple.Parent                 = parent
    ripple.Size                   = UDim2.new(0, 6, 0, 6)
    ripple.Position               = UDim2.new(0, x - 3, 0, y - 3)
    ripple.BackgroundColor3       = color or ICE.PURE
    ripple.BackgroundTransparency = 0.22
    ripple.BorderSizePixel        = 0
    ripple.ZIndex                 = 18
    newCorner(ripple, 999)
    local sz = 94
    TweenSvc:Create(ripple, TweenInfo.new(0.54, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size                   = UDim2.new(0, sz, 0, sz),
        Position               = UDim2.new(0, x - sz/2, 0, y - sz/2),
        BackgroundTransparency = 1.0,
    }):Play()
    local ring = Instance.new("Frame")
    ring.Parent                 = parent
    ring.Size                   = UDim2.new(0, 10, 0, 10)
    ring.Position               = UDim2.new(0, x - 5, 0, y - 5)
    ring.BackgroundTransparency = 1.0
    ring.BorderSizePixel        = 0
    ring.ZIndex                 = 17
    newCorner(ring, 999)
    local ringSt = newStroke(ring, 1.8, color or ICE.PURE, 0.0)
    local sz2 = 58
    TweenSvc:Create(ring, TweenInfo.new(0.42, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, sz2, 0, sz2), Position = UDim2.new(0, x - sz2/2, 0, y - sz2/2),
    }):Play()
    TweenSvc:Create(ringSt, TweenInfo.new(0.42, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Transparency = 1.0,
    }):Play()
    game:GetService("Debris"):AddItem(ripple, 0.60)
    game:GetService("Debris"):AddItem(ring,   0.50)
end

local burstChars = { "❄","·","✦","◆","❅","◇","*","❆","✧","·","•","❄" }

local function spawnFrostBurst(parent, color)
    local cx = BTN_W / 2 - 2
    local cy = BTN_H / 2 - 2
    for i = 1, 8 do
        local angle = (i - 1) * (math.pi * 2 / 8) + rnd(-0.28, 0.28)
        local dist  = rnd(16, 36)
        local tx    = cx + math.cos(angle) * dist
        local ty    = cy + math.sin(angle) * dist
        local shard = Instance.new("TextLabel")
        shard.Parent                 = parent
        shard.Size                   = UDim2.new(0, 11, 0, 11)
        shard.Position               = UDim2.new(0, cx - 5, 0, cy - 5)
        shard.Text                   = burstChars[math.random(#burstChars)]
        shard.TextSize               = math.random(6, 10)
        shard.Font                   = Enum.Font.GothamBold
        shard.BackgroundTransparency = 1
        shard.TextColor3             = color or ICE.PURE
        shard.TextTransparency       = 0
        shard.ZIndex                 = 19
        shard.Rotation               = math.random(0, 360)
        TweenSvc:Create(shard, TweenInfo.new(0.52, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Position         = UDim2.new(0, tx - 5, 0, ty - 5),
            TextTransparency = 1.0,
            Rotation         = shard.Rotation + math.random(90, 280),
        }):Play()
        game:GetService("Debris"):AddItem(shard, 0.60)
    end
end

-- ═══════════════════════════════════════════════════════════════
--  DRAG READY / HIDE
-- ═══════════════════════════════════════════════════════════════
local function showDragReady(btnData)
    tw(btnData.dragRingStroke, TW.FAST, { Transparency = 0.0 })
    tw(btnData.dragHint,       TW.FAST, { TextTransparency = 0.18 })
    spawnRipple(btnData.btn, BTN_W / 2, BTN_H / 2, ICE.DRAG_READY)
    -- лёгкое дрожание для тактильного фидбека
    local ox = btnData.root.Position.X.Offset
    local oy = btnData.root.Position.Y.Offset
    local sx = btnData.root.Position.X.Scale
    local sy = btnData.root.Position.Y.Scale
    tw(btnData.root, TW.WOBBLE, { Position = UDim2.new(sx, ox + 3, sy, oy) })
    delay(0.10, function()
        tw(btnData.root, TW.WOBBLE, { Position = UDim2.new(sx, ox - 3, sy, oy) })
    end)
    delay(0.20, function()
        tw(btnData.root, TW.BOUNCE, { Position = UDim2.new(sx, ox, sy, oy) })
    end)
end

local function hideDragReady(btnData)
    tw(btnData.dragRingStroke, TW.MED, { Transparency = 1.0 })
    tw(btnData.dragHint,       TW.MED, { TextTransparency = 1.0 })
    tw(btnData.dragProgressFill, TW.FAST, { Size = UDim2.new(0, 0, 0, 0) })
end

local pressState = false

local function onPressDown(btnData)
    if pressState then return end
    pressState = true
    tw(btnData.btn, TW.SNAP, { BackgroundTransparency = 0.22 })
end

local function onPressUp(btnData)
    if not pressState then return end
    pressState = false
    tw(btnData.btn, TW.SNAP, { BackgroundTransparency = 0 })
    hideDragReady(btnData)
end

-- ═══════════════════════════════════════════════════════════════
--  DRAG — 444 мс зажать → fill из центра → тащи куда угодно
--
--  Используем UIS.InputChanged (глобальное событие) чтобы
--  движение работало даже когда палец вышел за пределы кнопки.
-- ═══════════════════════════════════════════════════════════════
local function makeDraggable(btnData)
    local pressActive = false
    local dragActive  = false
    local moved       = false
    local dragStart   = nil
    local startPos    = nil
    local activeInput = nil

    -- Нажатие
    btnData.root.InputBegan:Connect(function(i)
        if i.UserInputType ~= Enum.UserInputType.Touch
        and i.UserInputType ~= Enum.UserInputType.MouseButton1 then return end

        activeInput = i
        pressActive = true
        dragActive  = false
        moved       = false
        dragStart   = Vector2.new(i.Position.X, i.Position.Y)
        startPos    = btnData.root.Position

        onPressDown(btnData)

        -- Fill из центра за DRAG_THRESHOLD секунд
        task.spawn(function()
            local t0fill = tick()
            while pressActive do
                local elapsed  = tick() - t0fill
                local progress = clamp(elapsed / DRAG_THRESHOLD, 0, 1)
                btnData.dragProgressFill.Size = UDim2.new(progress, 0, progress, 0)

                if elapsed >= DRAG_THRESHOLD then
                    dragActive = true
                    showDragReady(btnData)
                    break
                end
                task.wait(0.016)
            end
            -- отпустили раньше — схлопываем fill
            if not pressActive then
                btnData.dragProgressFill.Size = UDim2.new(0, 0, 0, 0)
            end
        end)
    end)

    -- Отпускание (глобально — работает даже вне кнопки)
    UIS.InputEnded:Connect(function(i)
        if i ~= activeInput then return end
        activeInput = nil
        pressActive = false
        dragActive  = false
        onPressUp(btnData)
    end)

    -- Движение (глобально — единственный рабочий способ на мобайле)
    UIS.InputChanged:Connect(function(i)
        if i ~= activeInput then return end
        if not dragActive then return end

        local dx = i.Position.X - dragStart.X
        local dy = i.Position.Y - dragStart.Y
        if math.sqrt(dx*dx + dy*dy) > 6 then moved = true end

        btnData.root.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + dx,
            startPos.Y.Scale, startPos.Y.Offset + dy
        )
    end)

    return function() return moved end
end

local aimMoved = makeDraggable(AIM)

-- ═══════════════════════════════════════════════════════════════
--  ТРЕКИНГ КАСАНИЯ ИГРОВОЙ ЗОНЫ
-- ═══════════════════════════════════════════════════════════════
local lastGameTouchPos = nil

local function isOnButton(pos)
    local abs = AIM.root.AbsolutePosition
    local sz  = AIM.root.AbsoluteSize
    return pos.X >= abs.X and pos.X <= abs.X + sz.X
       and pos.Y >= abs.Y and pos.Y <= abs.Y + sz.Y
end

UIS.TouchStarted:Connect(function(touch, _gp)
    local pos = Vector2.new(touch.Position.X, touch.Position.Y)
    if not isOnButton(pos) then
        lastGameTouchPos       = pos
        _G.AntiEmoLastTouchPos = pos
    end
end)

UIS.InputBegan:Connect(function(i, gp)
    if gp then return end
    if i.UserInputType == Enum.UserInputType.MouseButton1 then
        local pos = Vector2.new(i.Position.X, i.Position.Y)
        if not isOnButton(pos) then
            lastGameTouchPos       = pos
            _G.AntiEmoLastTouchPos = pos
        end
    end
end)

-- ═══════════════════════════════════════════════════════════════
--  ИГРОВАЯ ЛОГИКА
-- ═══════════════════════════════════════════════════════════════
local function isAlive(p)
    if not p.Character then return false end
    local h = p.Character:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

local function getNearestToScreenPos(screenPos)
    local best, bestDist = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and isAlive(p) then
            local root = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            if root then
                local vp, onScreen = Camera:WorldToViewportPoint(root.Position)
                if onScreen then
                    local d = (Vector2.new(vp.X, vp.Y) - screenPos).Magnitude
                    if d < bestDist then bestDist = d; best = p end
                end
            end
        end
    end
    return best
end

local function refreshAim()
    setAimVisual(API.isLocked())
end

-- Ripple при касании
AIM.btn.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.Touch
    or i.UserInputType == Enum.UserInputType.MouseButton1 then
        local abs = AIM.btn.AbsolutePosition
        spawnRipple(AIM.btn,
            i.Position.X - abs.X,
            i.Position.Y - abs.Y,
            aimVisualOn and ICE.MINT_HOT or ICE.PURE
        )
    end
end)

-- Основной обработчик нажатия
AIM.btn.MouseButton1Click:Connect(function()
    if aimMoved() then return end
    local touchPos = lastGameTouchPos or UIS:GetMouseLocation()
    local target   = getNearestToScreenPos(touchPos)
    if API.isLocked() then
        API.toggleAimLock()
    elseif target and API.lockOnPlayer then
        API.lockOnPlayer(target)
    else
        API.toggleAimLock()
    end
    refreshAim()
end)

-- Синхронизация состояния
task.spawn(function()
    while task.wait(0.4) do
        if not gui.Parent then break end
        refreshAim()
    end
end)
