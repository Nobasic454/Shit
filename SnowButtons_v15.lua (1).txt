local function rnd(a, b)       return a + math.random() * (b - a) end
local function lerp(a, b, t)   return a + (b - a) * t             end
local function clamp(v,mn,mx)  return math.max(mn, math.min(mx, v)) end
local function tw(obj, info, props) TweenSvc:Create(obj, info, props):Play() end
local function delay(t, fn)    task.delay(t, fn) end

local function newCorner(parent, r)
    local c = Instance.new("UICorner", parent); c.CornerRadius = UDim.new(0, r); return c
end
local function newStroke(parent, thick, col, trans)
    local s = Instance.new("UIStroke", parent)
    s.Thickness = thick or 2; s.Color = col or ICE.STROKE_IDLE; s.Transparency = trans or 0
    return s
end
local function newGrad(parent, c1, c2, rot)
    local g = Instance.new("UIGradient", parent)
    g.Color = ColorSequence.new{ColorSequenceKeypoint.new(0,c1), ColorSequenceKeypoint.new(1,c2)}
    g.Rotation = rot or 90; return g
end
local function newGrad3(parent, c1, c2, c3, rot)
    local g = Instance.new("UIGradient", parent)
    g.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, c1), ColorSequenceKeypoint.new(0.5, c2), ColorSequenceKeypoint.new(1, c3),
    }
    g.Rotation = rot or 90; return g
end

    for _ = 1, lcfg.count do
        local sf = Instance.new("TextLabel")
        sf.Parent                 = bgSnowFrame
        sf.Size                   = UDim2.new(0, 14, 0, 14)
        sf.Position               = UDim2.new(rnd(0, 1), 0, rnd(-0.05, 1.06), 0)
        sf.Text                   = BG_SNOW_CHARS[math.random(#BG_SNOW_CHARS)]
        sf.TextSize               = math.random(lcfg.sizeMin, lcfg.sizeMax)
        sf.Font                   = Enum.Font.Code
        sf.BackgroundTransparency = 1
        local rc = math.random(5)
        sf.TextColor3 = (rc == 1) and ICE.PURE
                     or (rc == 2) and ICE.FROST
                     or (rc == 3) and ICE.FLAKE
                     or (rc == 4) and ICE.WHITE
                     or ICE.SNOW_L1
        sf.TextTransparency       = rnd(lcfg.transMin, lcfg.transMax)
        sf.ZIndex                 = lcfg.zindex
        sf.Active                 = false
        bgSnowflakes[#bgSnowflakes + 1] = {
            label    = sf,
            speedY   = rnd(lcfg.speedMin, lcfg.speedMax),
            startX   = rnd(0, 1),
            wobble   = rnd(0, math.pi * 2),
            wSpeed   = rnd(0.10, 0.85) * lcfg.wobMult,
            rot      = rnd(0, 360),
            rotSpd   = rnd(-55, 55),
            transMin = lcfg.transMin,
            transMax = lcfg.transMax,
            sizeMin  = lcfg.sizeMin,
            sizeMax  = lcfg.sizeMax,

task.spawn(function()
    local time = 0
    while gui.Parent do
        local dt = task.wait(0.033)
        time = time + dt
        for _, sf in ipairs(bgSnowflakes) do
            local sl = sf.label
            local py = sl.Position.Y.Scale + sf.speedY * dt * 30
            local wobX = math.sin(time * sf.wSpeed + sf.wobble) * 0.007
            local px = sf.startX + wobX
            sf.rot = (sf.rot + sf.rotSpd * dt) % 360
            if py > 1.08 then
                py = rnd(-0.10, -0.01)
                px = rnd(0, 1)
                sf.startX = px
                sl.TextTransparency = rnd(sf.transMin, sf.transMax)
                sl.TextSize = math.random(sf.sizeMin, sf.sizeMax)
                sl.Text = BG_SNOW_CHARS[math.random(#BG_SNOW_CHARS)]
            end
            sl.Position = UDim2.new(clamp(px, 0, 1), 0, py, 0)
            sl.Rotation = sf.rot
        end
    end
end)

local function startParticles(btnData)
    task.spawn(function()
        local time = 0
        while gui.Parent do
            local dt = task.wait(0.033)
            time = time + dt
            for _, sf in ipairs(btnData.snowflakes) do
                local sl = sf.label
                local py = sl.Position.Y.Scale + sf.speedY * dt * 30
                local wobX = math.sin(time * sf.wSpeed + sf.wobble) * 0.014
                local px = sf.startX + wobX
                sf.rot = (sf.rot + sf.rotSpd * dt) % 360
                if py > 1.12 then
                    py = rnd(-0.15, -0.02)
                    px = rnd(0, 1.0)
                    sf.startX = px
                    sl.TextTransparency = rnd(0.25, 0.68)
                end
                sl.Position = UDim2.new(clamp(px, 0, 1.0), 0, py, 0)
                sl.Rotation = sf.rot
            end
        end
    end)
end

local function startCrystals(btnData)
    task.spawn(function()
        local time = 0
        while gui.Parent do
            time = time + task.wait(0.04)
            for i, cd in ipairs(btnData.crystals) do
                local phase = time * 0.62 + i * 1.1
                local s = 0.5 + 0.5 * math.sin(phase)
                cd.label.TextTransparency = lerp(cd.baseTrans, cd.baseTrans - 0.32, s)
                cd.label.Rotation = cd.baseRot + math.sin(phase * 0.8) * 13
                cd.label.TextSize = math.max(4, math.floor(cd.baseSize * lerp(0.88, 1.0, s) + 0.5))
            end
        end
    end)
end

--  DRAGGABLE
-- ═══════════════════════════════════════════════════════════════

local function makeDraggable(btnData)
    local drag, ds, sp = false, nil, nil
    local moved      = false
    local holdActive = false

    btnData.btn.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch
        or i.UserInputType == Enum.UserInputType.MouseButton1 then
            drag = true; ds = i.Position; sp = btnData.root.Position
            moved = false; holdActive = true
            onPressDown(btnData)
            task.delay(DRAG_THRESHOLD, function()
                if holdActive then showDragReady(btnData) end
            end)
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then
                    drag = false; holdActive = false
                    onPressUp(btnData)
                end
            end)
        end
    end)

    btnData.btn.InputChanged:Connect(function(i)
        if drag and (i.UserInputType == Enum.UserInputType.Touch
                  or i.UserInputType == Enum.UserInputType.MouseMovement) then
            local d = i.Position - ds
            if d.Magnitude > 8 then moved = true end
            btnData.root.Position = UDim2.new(sp.X.Scale, sp.X.Offset+d.X, sp.Y.Scale, sp.Y.Offset+d.Y)
        end
    end)

    return function() return moved end
end

-- ═══════════════════════════════════════════════════════════════
--  TOUCH TRACKING
-- ═══════════════════════════════════════════════════════════════

local lastGameTouchPos = nil

local function isOnButton(pos)
    local abs = AIM.root.AbsolutePosition; local sz = AIM.root.AbsoluteSize
    return pos.X >= abs.X and pos.X <= abs.X+sz.X and pos.Y >= abs.Y and pos.Y <= abs.Y+sz.Y
end

UIS.TouchStarted:Connect(function(touch, _gp)
    local pos = Vector2.new(touch.Position.X, touch.Position.Y)
    if not isOnButton(pos) then lastGameTouchPos = pos; _G.AntiEmoLastTouchPos = pos end
end)
UIS.InputBegan:Connect(function(i, gp)
    if gp then return end
    if i.UserInputType == Enum.UserInputType.MouseButton1 then
        local pos = Vector2.new(i.Position.X, i.Position.Y)
        if not isOnButton(pos) then lastGameTouchPos = pos; _G.AntiEmoLastTouchPos = pos end
    end
end)

--  PLAYER UTILS
-- ═══════════════════════════════════════════════════════════════

local function isAlive(p)
    if not p.Character then return false end
    local h = p.Character:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

local function getNearestToScreenPos(screenPos)
    local best, bestDist = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and isAlive(p) then
            local playerRoot = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            if playerRoot then
                local vp, onScreen = Camera:WorldToViewportPoint(playerRoot.Position)
                if onScreen then
                    local d = (Vector2.new(vp.X,vp.Y) - screenPos).Magnitude
                    if d < bestDist then bestDist = d; best = p end
                end
            end
        end
    end
    return best
end

local function refreshAim() setAimVisual(API.isLocked()) end

local aimMoved = makeDraggable(AIM)

AIM.btn.MouseButton1Click:Connect(function()
    if aimMoved() then return end
    local touchPos = lastGameTouchPos or UIS:GetMouseLocation()
    local target   = getNearestToScreenPos(touchPos)
    spawnSuperTouch(AIM.btn, BTN_W/2, BTN_H/2, aimVisualOn and ICE.MINT_HOT or ICE.PURE)
    if API.isLocked() then
        API.toggleAimLock()
    elseif target and API.lockOnPlayer then
        API.lockOnPlayer(target)
    else
        API.toggleAimLock()
    end
    refreshAim()
end)

task.spawn(function()
    while task.wait(0.4) do
        if not gui.Parent then break end
        refreshAim()
    end
end)
